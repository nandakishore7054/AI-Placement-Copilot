1. V1 → V2 Change Summary
#	Refinement	V1 State	V2 State	Impact
1	CompanyMember RBAC	User.companyId + implicit "owner" via Company.ownerId	New CompanyMember join model with CompanyRole enum (OWNER, ADMIN, RECRUITER, INTERVIEWER, HR)	1 new model, 1 new enum, User.companyId removed
2	InterviewQuestion model	Interview.questions String[]	New InterviewQuestion model with questionText, topic, difficulty, orderIndex	1 new model, 1 new enum, replaces String[]
3	Structured Resume analytics	Resume.aiAnalysis Json? + Resume.score Int?	New ResumeAnalysis model with typed fields: overallScore, formatScore, contentScore, keywordsFound[], keywordsMissing[], suggestions[], sectionScores Json, rawAnalysis Json?	1 new model, structured fields replace opaque JSON
4	AuditLog model	Not present	New AuditLog model tracking action, entityType, entityId, userId, metadata, ipAddress	1 new model, 1 new enum
5	AI Intelligence Layer	Not present	4 new models: SkillGap, Recommendation, CareerInsight, CareerRoadmap + 3 new enums	4 new models, 3 new enums
6	StudentProfile model	Student data embedded in User	New StudentProfile with bio, skills[], education, preferredLocations[], preferredCategories[], linkedinUrl, githubUrl, portfolioUrl, expectedSalary	1 new model, student-specific data decoupled from User
7	Updated relationships	See V1	All FK relationships updated for new models	Cascading deletes, new indexes
8	Updated indexes	Basic indexes	Compound indexes, covering indexes for new models	Performance optimization
9	Server Action vs API Route	Mixed responsibilities	Clear separation: Server Actions for all authenticated mutations; API Routes only for webhooks, cron, Vapi callbacks, and public read APIs	Cleaner architecture
10	Everything else	V1 baseline	Unchanged	No regression

// ════════════════════════════════════════════════════════════════
// USER & AUTH (Synced via Clerk Webhooks)
// ════════════════════════════════════════════════════════════════
model User {
  id              String    @id                  // Clerk user ID
  email           String    @unique
  firstName       String
  lastName        String
  imageUrl        String?
  role            UserRole  @default(STUDENT)
  onboardingDone  Boolean   @default(false)
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt
  // Student relations
  studentProfile  StudentProfile?
  resume          Resume?
  applications    Application[]
  interviews      Interview[]
  feedbacks       Feedback[]
  subscription    Subscription?
  skillGaps       SkillGap[]
  recommendations Recommendation[]
  careerRoadmaps  CareerRoadmap[]
  // Company relations (via CompanyMember join)
  companyMemberships CompanyMember[]
  // Audit trail
  auditLogs       AuditLog[]
  @@index([email])
  @@index([role])
}
// ════════════════════════════════════════════════════════════════
// STUDENT PROFILE (Decoupled student-specific data)
// ════════════════════════════════════════════════════════════════
model StudentProfile {
  id                    String    @id @default(cuid())
  bio                   String?   @db.Text
  skills                String[]                          // ["React", "TypeScript", "Node.js"]
  education             String?   @db.Text                // JSON string or free text
  preferredLocations    String[]                          // ["Bangalore", "Remote"]
  preferredCategories   String[]                          // ["Software Developer", "Data Science"]
  expectedSalary        String?                           // "5-8 LPA"
  linkedinUrl           String?
  githubUrl             String?
  portfolioUrl          String?
  yearsOfExperience     Int?
  createdAt             DateTime  @default(now())
  updatedAt             DateTime  @updatedAt
  // Relations
  user                  User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId                String    @unique
  @@index([skills], type: Gin)
  @@index([preferredLocations], type: Gin)
}
// ════════════════════════════════════════════════════════════════
// COMPANY (Recruiter Organization)
// ════════════════════════════════════════════════════════════════
model Company {
  id          String   @id @default(cuid())
  name        String
  email       String   @unique
  logoUrl     String?                            // Cloudinary URL
  website     String?
  description String?  @db.Text
  industry    String?
  size        String?                            // "1-10", "11-50", "51-200", etc.
  verified    Boolean  @default(false)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  // Relations
  members     CompanyMember[]
  jobs        Job[]
  experiences Experience[]
  @@index([name])
  @@index([verified])
}
// ════════════════════════════════════════════════════════════════
// COMPANY MEMBER (RBAC Join Table)
// ════════════════════════════════════════════════════════════════
model CompanyMember {
  id        String      @id @default(cuid())
  role      CompanyRole @default(RECRUITER)
  joinedAt  DateTime    @default(now())
  updatedAt DateTime    @updatedAt
  // Relations
  user      User        @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId    String
  company   Company     @relation(fields: [companyId], references: [id], onDelete: Cascade)
  companyId String
  @@unique([userId, companyId])              // One membership per user per company
  @@index([companyId, role])                 // Find all recruiters in a company
  @@index([userId])                          // Find all companies for a user
}
// ════════════════════════════════════════════════════════════════
// RESUME (Student Resume + File Storage)
// ════════════════════════════════════════════════════════════════
model Resume {
  id              String                          @id @default(cuid())
  fileUrl         String                          // Cloudinary URL
  fileName        String
  extractedText   String?                         @db.Text
  embedding       Unsupported("vector(768)")?     // pgvector: Gemini text-embedding-004
  createdAt       DateTime                        @default(now())
  updatedAt       DateTime                        @updatedAt
  // Relations
  user            User                            @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId          String                          @unique
  analysis        ResumeAnalysis?
}
// ════════════════════════════════════════════════════════════════
// RESUME ANALYSIS (Structured AI Analysis — replaces aiAnalysis Json)
// ════════════════════════════════════════════════════════════════
model ResumeAnalysis {
  id                String    @id @default(cuid())
  overallScore      Int                               // 0-100
  formatScore       Int                               // 0-100 (layout, readability, length)
  contentScore      Int                               // 0-100 (relevance, depth, impact)
  atsScore          Int                               // 0-100 (ATS-friendliness)
  keywordsFound     String[]                          // Keywords matching industry standards
  keywordsMissing   String[]                          // Recommended keywords not present
  suggestions       String[]                          // Actionable improvement suggestions
  sectionScores     Json                              // { "experience": 85, "education": 70, "skills": 90, ... }
  strengths         String[]                          // Top strengths identified
  weaknesses        String[]                          // Areas needing improvement
  rawAnalysis       Json?                             // Full AI response for debugging/future use
  analyzedAt        DateTime  @default(now())
  updatedAt         DateTime  @updatedAt
  // Relations
  resume            Resume    @relation(fields: [resumeId], references: [id], onDelete: Cascade)
  resumeId          String    @unique
  @@index([overallScore])
}
// ════════════════════════════════════════════════════════════════
// JOB LISTINGS
// ════════════════════════════════════════════════════════════════
model Job {
  id            String                          @id @default(cuid())
  title         String
  description   String                          @db.Text      // HTML (rich text)
  location      String
  category      String
  level         JobLevel
  type          JobType                         @default(FULL_TIME)
  salary        String?                         // "5-8 LPA" or "50,000 INR/month"
  applyLink     String?                         // External application URL
  isVisible     Boolean                         @default(true)
  embedding     Unsupported("vector(768)")?     // pgvector: job title+description embedding
  createdAt     DateTime                        @default(now())
  updatedAt     DateTime                        @updatedAt
  // Relations
  company       Company                         @relation(fields: [companyId], references: [id], onDelete: Cascade)
  companyId     String
  applications  Application[]
  @@index([category])
  @@index([level])
  @@index([location])
  @@index([isVisible])
  @@index([createdAt(sort: Desc)])
  @@index([companyId])
}
// ════════════════════════════════════════════════════════════════
// JOB APPLICATIONS
// ════════════════════════════════════════════════════════════════
model Application {
  id          String              @id @default(cuid())
  status      ApplicationStatus   @default(PENDING)
  appliedAt   DateTime            @default(now())
  updatedAt   DateTime            @updatedAt
  notes       String?             @db.Text
  // Relations
  user        User                @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId      String
  job         Job                 @relation(fields: [jobId], references: [id], onDelete: Cascade)
  jobId       String
  @@unique([userId, jobId])                   // Prevent duplicate applications
  @@index([status])
  @@index([appliedAt(sort: Desc)])
  @@index([userId, status])                   // Student's applications by status
  @@index([jobId, status])                    // Job's applicants by status
}
// ════════════════════════════════════════════════════════════════
// AI MOCK INTERVIEWS
// ════════════════════════════════════════════════════════════════
model Interview {
  id            String            @id @default(cuid())
  role          String                            // Target job role
  type          InterviewType
  level         JobLevel
  techStack     String[]                          // Array of technologies
  status        InterviewStatus   @default(DRAFT)
  coverImage    String?                           // Random cover image path
  transcript    String?           @db.Text        // Full interview transcript
  duration      Int?                              // Duration in seconds
  createdAt     DateTime          @default(now())
  updatedAt     DateTime          @updatedAt
  // Relations
  user          User              @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId        String
  questions     InterviewQuestion[]               // ← V2: replaces String[]
  feedback      Feedback?
  @@index([userId])
  @@index([status])
  @@index([createdAt(sort: Desc)])
  @@index([userId, status])                       // User's interviews by status
}
// ════════════════════════════════════════════════════════════════
// INTERVIEW QUESTIONS (V2: Replaces String[] on Interview)
// ════════════════════════════════════════════════════════════════
model InterviewQuestion {
  id              String              @id @default(cuid())
  questionText    String              @db.Text
  topic           String?                               // "React Hooks", "System Design", etc.
  difficulty      QuestionDifficulty  @default(MEDIUM)
  orderIndex      Int                                   // Display order (0-based)
  expectedAnswer  String?             @db.Text           // AI-generated ideal answer hint
  createdAt       DateTime            @default(now())
  // Relations
  interview       Interview           @relation(fields: [interviewId], references: [id], onDelete: Cascade)
  interviewId     String
  @@index([interviewId, orderIndex])                    // Ordered questions for interview
  @@index([topic])
  @@index([difficulty])
}
// ════════════════════════════════════════════════════════════════
// INTERVIEW FEEDBACK (AI-Generated)
// ════════════════════════════════════════════════════════════════
model Feedback {
  id                    String    @id @default(cuid())
  totalScore            Int                               // 0-100
  categoryScores        Json                              // Array<{ name, score, comment }>
  strengths             String[]
  areasForImprovement   String[]
  finalAssessment       String    @db.Text
  questionsAnalysis     Json                              // Array<{ questionId, response, score, suggestion }>
  createdAt             DateTime  @default(now())
  // Relations
  interview             Interview @relation(fields: [interviewId], references: [id], onDelete: Cascade)
  interviewId           String    @unique
  user                  User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId                String
  @@index([userId])
  @@index([totalScore])
  @@index([userId, createdAt(sort: Desc)])               // User's feedback history
}
// ════════════════════════════════════════════════════════════════

  level       JobLevel

  imageUrl    String?                             // Company/role image
  isVisible   Boolean   @default(true)
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
  // Relations
  company     Company   @relation(fields: [companyId], references: [id], onDelete: Cascade)
  companyId   String
  @@index([category])
  @@index([isVisible])
  @@index([createdAt(sort: Desc)])
  @@index([companyId])
}
// ════════════════════════════════════════════════════════════════

model Subscription {
  id            String    @id @default(cuid())
  email         String    @unique
  isActive      Boolean   @default(true)
  subscribedAt  DateTime  @default(now())
  // Optional user link
  user          User?     @relation(fields: [userId], references: [id], onDelete: SetNull)
  userId        String?   @unique
  @@index([isActive])
}
// ════════════════════════════════════════════════════════════════
// AUDIT LOG (V2: New — Compliance & Debugging)
// ════════════════════════════════════════════════════════════════
model AuditLog {
  id          String       @id @default(cuid())
  action      AuditAction
  entityType  AuditEntity
  entityId    String                              // ID of the affected entity
  metadata    Json?                               // Additional context (old value, new value, etc.)
  ipAddress   String?
  userAgent   String?
  createdAt   DateTime     @default(now())
  // Relations
  user        User?        @relation(fields: [userId], references: [id], onDelete: SetNull)
  userId      String?
  @@index([entityType, entityId])                 // Find all logs for an entity
  @@index([userId, createdAt(sort: Desc)])        // User's activity timeline
  @@index([action])
  @@index([createdAt(sort: Desc)])                // Global activity feed
}
// ════════════════════════════════════════════════════════════════
// AI INTELLIGENCE LAYER — SKILL GAP ANALYSIS
// ════════════════════════════════════════════════════════════════
model SkillGap {
  id                String    @id @default(cuid())
  targetRole        String                            // Role the user aspires to
  currentSkills     String[]                          // Skills user currently has
  requiredSkills    String[]                          // Skills required for target role
  missingSkills     String[]                          // Gap: required - current
  proficiencyMap    Json                              // { "React": 80, "TypeScript": 60, ... }
  matchPercentage   Int                               // 0-100 overall match
  recommendations   String[]                          // Suggested learning resources
  analyzedAt        DateTime  @default(now())
  updatedAt         DateTime  @updatedAt
  // Relations
  user              User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId            String
  @@index([userId])
  @@index([targetRole])
  @@index([matchPercentage])
  @@index([userId, analyzedAt(sort: Desc)])
}
// ════════════════════════════════════════════════════════════════
// AI INTELLIGENCE LAYER — RECOMMENDATIONS
// ════════════════════════════════════════════════════════════════
model Recommendation {
  id              String              @id @default(cuid())
  type            RecommendationType
  title           String
  description     String              @db.Text
  relevanceScore  Float                                 // 0.0-1.0 (AI confidence)
  metadata        Json?                                 // Type-specific data (jobId, courseUrl, etc.)
  isActioned      Boolean             @default(false)   // User clicked/followed
  isDismissed     Boolean             @default(false)   // User dismissed
  createdAt       DateTime            @default(now())
  expiresAt       DateTime?                             // Optional TTL
  // Relations
  user            User                @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId          String
  @@index([userId, type])
  @@index([userId, isActioned, isDismissed])            // Active recommendations
  @@index([relevanceScore(sort: Desc)])
  @@index([createdAt(sort: Desc)])
}
// ════════════════════════════════════════════════════════════════
// AI INTELLIGENCE LAYER — CAREER INSIGHTS
// ════════════════════════════════════════════════════════════════
model CareerInsight {
  id            String           @id @default(cuid())
  category      InsightCategory
  title         String
  content       String           @db.Text                // Markdown-formatted insight
  dataPoints    Json?                                    // Supporting statistics/data
  source        String?                                  // Where the insight was derived from
  isPublished   Boolean          @default(true)
  createdAt     DateTime         @default(now())
  updatedAt     DateTime         @updatedAt
  @@index([category])
  @@index([isPublished, createdAt(sort: Desc)])
}
// ════════════════════════════════════════════════════════════════
// AI INTELLIGENCE LAYER — CAREER ROADMAP
// ════════════════════════════════════════════════════════════════
model CareerRoadmap {
  id              String    @id @default(cuid())
  targetRole      String                              // "Senior Frontend Engineer"
  currentLevel    JobLevel                            // User's current level
  targetLevel     JobLevel                            // Desired level
  timelineMonths  Int                                 // Estimated months to reach target
  milestones      Json                                // Array<{ title, description, skills, month, completed }>
  skillsToAcquire String[]                            // Skills needed along the way
  resources       Json?                               // Array<{ title, url, type }> learning resources
  progress        Int       @default(0)               // 0-100 completion percentage
  isActive        Boolean   @default(true)
  generatedAt     DateTime  @default(now())
  updatedAt       DateTime  @updatedAt
  // Relations
  user            User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId          String
  @@index([userId, isActive])
  @@index([targetRole])
  @@index([userId, generatedAt(sort: Desc)])
}

6. Updated API Design
Architecture Principle (V2 — Clarified)
Server Actions: All authenticated mutations and authenticated reads. Called from Client/Server Components.

API Routes: Only for (1) webhook receivers, (2) cron jobs, (3) Vapi callbacks, (4) public unauthenticated reads, and (5) the semantic search endpoint which requires raw SQL.
